const prayers = [ { id: 'fajr', bn: 'ফজর', en: 'Fajr' }, { id: 'dhuhr', bn: 'জোহর', en: 'Dhuhr' }, { id: 'asr', bn: 'আসর', en: 'Asr' }, { id: 'maghrib', bn: 'মাগরিব', en: 'Maghrib' }, { id: 'isha', bn: 'এশা', en: 'Isha' } ]; // পারমিশন দেওয়া ইমেইল ও পাসওয়ার্ড const allowedUsers = { "minhaj@gmail.com": "gando", // মিনহাজের লগইন "nadiya@gmail.com": "gando" // নাদিয়ার লগইন }; let currentLang = 'bn'; function handleLogin() { const emailInput = document.getElementById('email').value.trim(); const passwordInput = document.getElementById('password').value.trim(); if (allowedUsers[emailInput] && allowedUsers[emailInput] === passwordInput) { document.getElementById('login-modal').classList.add('hidden'); document.getElementById('dashboard').classList.remove('hidden'); document.getElementById('user-display-name').innerText = emailInput.split('@')[0]; renderPrayers(); } else { alert('ভুল ইমেইল অথবা পাসওয়ার্ড! সঠিক তথ্য দিয়ে চেষ্টা করুন।'); } } function toggleTheme() { document.body.classList.toggle('dark-mode'); } function toggleLanguage() { currentLang = currentLang === 'bn' ? 'en' : 'bn'; renderPrayers(); } function renderPrayers() { const users = ['minhaj', 'nadiya']; users.forEach(user => { const container = document.getElementById(`${user}-prayers`); if (!container) return; container.innerHTML = ''; prayers.forEach(p => { const name = currentLang === 'bn' ? p.bn : p.en; const div = document.createElement('div'); div.style.cssText = 'display:flex; justify-content:space-between; align-items:center; margin:10px 0; padding:8px; border-bottom:1px solid var(--border-color);'; div.innerHTML = ` ${name} `; container.appendChild(div); }); }); }
